== rustup_channel and expand_version, model.rs
    /// The rustup channel this row names:
    /// the declaration's `packageId` when it has one,
    /// else the locked version.
    /// Why:
    /// `rustup` names a toolchain by the channel it was installed as (`stable`,
    /// `nightly-2026-10-30`),
    /// and `rustup toolchain list` prints that channel followed by the host triple,
    /// so every stage that names this row's toolchain (its install,
    /// its components and targets,
    /// its listing,
    /// its observation) asks for the same text,
    /// never for the release the channel resolved to.
    pub fn rustup_channel(&self) -> &str {
        return self
            .declaration
            .package_id
            .as_deref()
            .unwrap_or(&self.version);
    }
}

//endregion Declaration and lock entry

/// Re-exports the decoding functions,
pub fn expand_version(
    declaration: &Declaration,
    version: &str,
) -> Declaration {
    let mut expanded: Declaration = declaration.clone();
    if let Some(text) = &declaration.package_id {
        expanded.package_id = Some(text.replace(VERSION_PLACEHOLDER, version));
    }
    let mut env: Vec<(String, String)> = Vec::new();
    for (name, text) in &declaration.env {
        env.push((name.clone(), text.replace(VERSION_PLACEHOLDER, version)));
    }
    expanded.env = env;
    return expanded;
}
