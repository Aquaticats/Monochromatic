## closure_driver_probe.rs lines 150-193 at the pinned commit: active_toolchain removes RUSTUP_TOOLCHAIN, CARGO and RUSTC before asking rustup
150: 
151: /// What:
152: ///  The first word of the output of `rustup show active-toolchain` in a directory.
153: /// Why:
154: ///  That word names the driver directory.
155: ///  The environment is cleaned as `cargo-dylint` cleans it,
156: ///  because an inherited `RUSTUP_TOOLCHAIN` is not what it asks.
157: ///
158: /// In TS you'd write (pseudocode):
159: /// ```ts
160: /// function activeToolchain(rustup: Path, directory: Path): string; // throws string
161: /// ```
162: pub(crate) fn active_toolchain(
163:     rustup: &Path,
164:     directory: &Path,
165: ) -> Result<String, String> {
166:     let output = match Command::new(rustup)
167:         .args(["show", "active-toolchain"])
168:         .current_dir(directory)
169:         .env_remove("RUSTUP_TOOLCHAIN")
170:         .env_remove("CARGO")
171:         .env_remove("RUSTC")
172:         .stdin(Stdio::null())
173:         .output()
174:     {
175:         Ok(done) => done,
176:         Err(error) => return Err(format!("rustup does not run ({error})")),
177:     };
178:     if !output.status.success() {
179:         return Err(format!(
180:             "`rustup show active-toolchain` failed ({})",
181:             output.status
182:         ));
183:     }
184:     return match String::from_utf8_lossy(&output.stdout)
185:         .split_ascii_whitespace()
186:         .next()
187:     {
188:         Some(word) => Ok(word.to_string()),
189:         None => Err(String::from(
190:             "`rustup show active-toolchain` printed no toolchain",
191:         )),
192:     };
193: }
