pub const VERSION_PLACEHOLDER: &str = "{version}";

/// What:
/// A copy of the declaration with `{version}` replaced by `version` in `packageId` and in every `env` value.
/// Why:
/// A Rust channel's package id (`nightly-2026-10-30`) and a tool's home variable (`android-sdk/23.0`) equal the
///           resolved version,
/// which changes on every update.
/// The plan writes `{version}` once,
/// the lock holds the
///           expanded text,
/// and comparing a plan with a lock expands the plan with the lock's version first,
/// so an
///           update never makes the two files disagree.
///
/// In TS you'd write (pseudocode):
/// ```ts
/// export function expandVersion(declaration: Declaration, version: string): Declaration
/// ```
pub fn expand_version(
    declaration: &Declaration,
    version: &str,
) -> Declaration {
    let mut expanded: Declaration = declaration.clone();
    if let Some(text) = &declaration.package_id {
        expanded.package_id = Some(text.replace(VERSION_PLACEHOLDER, version));
    }
    let mut env: Vec<(String, String)> = Vec::new();
    for (name, text) in &declaration.env {
        env.push((name.clone(), text.replace(VERSION_PLACEHOLDER, version)));

## install_rust_android.rs, install_rust_toolchain and install_rust_components (lines 56-139)
/// `rustup toolchain install --no-self-update <package id, or the locked version>`.
pub fn install_rust_toolchain(
    ctx: &Context,
    tool: &LockedTool,
) -> Result<(), ToolchainError> {
    let channel: &str = tool.rustup_channel();
    let request: RunRequest = ctx.request(
        "rustup",
        &["toolchain", "install", "--no-self-update", channel],
    );
    run_checked(
        ctx,
        &request,
        "rustup toolchain install",
        &tool.declaration.id,
    )?;
    return Ok(());
}

/// What:
/// `rustup component add <name> --toolchain <version>` for each declared component,
/// then `rustup target add
///           <triple> --toolchain <version>` for each declared target.
/// Why:
/// The toolchain install above brings the default profile;
/// the plan names the extras.
///
/// In TS you'd write (pseudocode):
/// ```ts
/// export function installRustComponents(ctx: Context, tool: LockedTool): void
/// ```
pub fn install_rust_components(
    ctx: &Context,
    tool: &LockedTool,
) -> Result<(), ToolchainError> {
    for component in tool
        .declaration
        .options
        .rust_components
        .clone()
        .unwrap_or_default()
    {
        let request: RunRequest = ctx.request(
            "rustup",
            &[
                "component",
                "add",
                &component,
                "--toolchain",
                tool.rustup_channel(),
            ],
        );
        run_checked(
            ctx,
            &request,
            &format!("rustup component add ({component})"),
            &tool.declaration.id,
        )?;
    }
    for target in tool
        .declaration
        .options
        .rust_targets
        .clone()
        .unwrap_or_default()
    {
        let request: RunRequest = ctx.request(
            "rustup",
            &[
                "target",
                "add",
                &target,
                "--toolchain",
                tool.rustup_channel(),
            ],
        );
        run_checked(
            ctx,
            &request,
            &format!("rustup target add ({target})"),
            &tool.declaration.id,
        )?;
    }
    return Ok(());

/// Components and targets are added to the channel `packageId` names:
/// that is the toolchain the install step created,
/// and the locked version can spell a date the channel does not.
#[test]
fn rust_components_and_targets_use_the_package_id_channel_that_was_installed() {
    let mut rig = Rig::new();
    let rust = rig
        .lock
        .tools
        .iter_mut()
        .find(|tool| return tool.declaration.id == "rust")
        .expect("the rust tool");
    rust.declaration.package_id = Some("nightly".to_string());
    install_tool(&rig.context(), &rig.tool("rust")).expect("installed");
    assert_eq!(
        rig.runner.lines(),
        vec![
            "rustup toolchain install --no-self-update nightly".to_string(),
            "rustup component add clippy --toolchain nightly".to_string(),
            "rustup component add rust-src --toolchain nightly".to_string(),
            "rustup target add aarch64-linux-android --toolchain nightly".to_string(),
        ]
    );
